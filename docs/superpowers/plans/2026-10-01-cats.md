# Cats Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add three animated pixel-art cats (Misa, Max, Raye) that perch around the home, lobby and gameplay screens, react to the game and to the mouse, and act out their personalities.

**Architecture:** Client-only. Sprites are 16×16 text grids recoloured per cat and pre-rendered to small canvases. Game cues come from a pure function that diffs consecutive `GameState`s. Each cat's behaviour is a pure state machine `(state, input, dtMs, rng) → state`. A single `CatLayer` overlay runs one `requestAnimationFrame` loop, positions each cat relative to anchored DOM elements, and draws only when a cat's frame changes.

**Tech Stack:** TypeScript (strict), React 19, Vite, canvas 2D, Vitest, Playwright.

**Spec:** `docs/superpowers/specs/2026-10-01-cats-design.md`

```ts
import { PALETTES, SPRITES, SPRITE_SIZE, type Grid, type Palette } from './sprites';
import type { CatName, Pose } from './types';

export interface Pixel {
  x: number;
  y: number;
  color: string;
}

export function framePixels(grid: Grid, palette: Palette): Pixel[] {
  const pixels: Pixel[] = [];
  grid.forEach((row, y) =>
    [...row].forEach((ch, x) => {
      if (ch !== '.') pixels.push({ x, y, color: palette[ch] });
    }),
  );
  return pixels;
}

export type Sheet = Record<Pose, HTMLCanvasElement[]>;

/** Draws every pose frame of one cat once, at 1 px per sprite pixel (CSS scales it up). */
export function buildSheet(cat: CatName): Sheet {
  const sheet = {} as Sheet;
  for (const pose of Object.keys(SPRITES) as Pose[]) {
    sheet[pose] = SPRITES[pose].map((grid) => {
      const canvas = document.createElement('canvas');
      canvas.width = SPRITE_SIZE;
      canvas.height = SPRITE_SIZE;
      const ctx = canvas.getContext('2d')!;
      for (const { x, y, color } of framePixels(grid, PALETTES[cat])) {
        ctx.fillStyle = color;
        ctx.fillRect(x, y, 1, 1);
      }
      return canvas;
    });
  }
  return sheet;
}
```

```ts
  sit: 600,

  pounce: 1000,

  peek: 1000,

  alert: 1000,
  stretch: 500,
  yawn: 1000,
  earTwitch: 1000,
  oneEye: 1000,

export function frameIndex(pose: Pose, poseMs: number): number {
  return Math.floor(poseMs / FRAME_MS[pose]) % SPRITES[pose].length;
}

export function between(rng: Rng, min: number, max: number): number {
  return min + rng() * (max - min);
```

```ts
import {
  BOARD_HEIGHT,
  HIDDEN_ROWS,
  ghostPiece,
  pieceCells,
  type ActivePiece,
  type GameState,
  type PlayerId,
} from '@mt/engine';

export type CatEvent =
  | { type: 'clear'; count: number }
  | { type: 'hardDrop'; player: PlayerId; column: number };

/** Gravity and soft drop never move a piece this far between two consecutive states. */
export const HARD_DROP_MIN_ROWS = 3;
/** Danger turns on when a locked cell is in the top N visible rows... */
export const DANGER_ON_ROWS = 6;
/** ...and off only once no locked cell is in the top N visible rows. */
export const DANGER_OFF_ROWS = 8;

export interface EventTracker {
  prev: GameState | null;
  /** Highest lock count and line total seen, so reconciled snapshots never replay events. */
  lockCount: number;
  lines: number;
}

export const initialTracker: EventTracker = { prev: null, lockCount: 0, lines: 0 };

export function trackGame(
  tracker: EventTracker,
  next: GameState | null,
): { tracker: EventTracker; events: CatEvent[] } {
  if (!next) return { tracker: initialTracker, events: [] };
  const prev = tracker.prev;
  if (!prev || next.tick < prev.tick) {
    // First state of a game, or a new game after Play again: baseline only.
    return { tracker: { prev: next, lockCount: next.lockCount, lines: next.lines }, events: [] };
  }

  const events: CatEvent[] = [];
  if (next.lockCount > tracker.lockCount) {
    for (const player of [0, 1] as const) {
      const column = hardDropColumn(prev, next, player);
      if (column !== null) events.push({ type: 'hardDrop', player, column });
    }
  }
  if (next.lines > tracker.lines) {
    events.push({ type: 'clear', count: next.lastClear?.count ?? next.lines - tracker.lines });
  }
  return {
    tracker: {
      prev: next,
      lockCount: Math.max(tracker.lockCount, next.lockCount),
      lines: Math.max(tracker.lines, next.lines),
    },
    events,
  };
}

/** Landing column if `player` hard-dropped between the two states, else null. */
function hardDropColumn(prev: GameState, next: GameState, player: PlayerId): number | null {
  const before = prev.players[player];
  const after = next.players[player];
  if (!before.active) return null;
  // A lock always takes the next piece from the queue; a hold that takes from the queue also fills the hold slot.
  const locked = before.queue.join() !== after.queue.join() && before.hold === after.hold;
  if (!locked) return null;
  const landing = ghostPiece(prev, player);
  if (!landing || landing.y - before.active.y < HARD_DROP_MIN_ROWS) return null;
  return centerColumn(landing);
}

export function centerColumn(piece: ActivePiece): number {
  const xs = pieceCells(piece).map(([x]) => x);
  return (Math.min(...xs) + Math.max(...xs) + 1) / 2;
}

/** Rows from the top of the visible well to the highest locked cell (negative if in the hidden rows). */
function stackDepth(state: GameState): number | null {
  for (let y = 0; y < BOARD_HEIGHT; y++) {
    if (state.board[y].some((cell) => cell !== null)) return y - HIDDEN_ROWS;
  }
  return null;
}

export function dangerLevel(state: GameState | null, wasOn: boolean): boolean {
  if (!state) return false;
  const depth = stackDepth(state);
  if (depth === null || depth >= DANGER_OFF_ROWS) return false;
  if (depth < DANGER_ON_ROWS) return true;
  return wasOn;
}
```

```ts
import { between } from './anim';
import type { Bubble, CatView, Rng } from './types';

export interface MaxInput {
  /** Length of Max's edge in px; positions run from 0 (left end) to width. */
  width: number;
  /** Where Max wants to stand, or null to wander. */
  target: number | null;
  /** Pounce toward this position this step (a hard drop landed there). */
  pounceAt: number | null;
  /** A line was cleared this step. */
  meow: boolean;
  reducedMotion: boolean;
}

type MaxAction = 'sit' | 'walk' | 'run' | 'crouch' | 'pounce' | 'stretch';

export interface MaxState {
  action: MaxAction;
  actionMs: number;
  x: number;
  y: number;
  facing: 1 | -1;
  jumpFrom: number;
  jumpTo: number;
  wanderTarget: number | null;
  wanderMs: number;
  meowMs: number;
  stretchMs: number;
  bubble: Bubble | null;
  bubbleMs: number;
  bubbleCount: number;
}

/** Speeds in px per ms. */
export const MAX_WALK_SPEED = 0.08;
export const MAX_RUN_SPEED = 0.25;
export const MAX_CROUCH_MS = 150;
export const MAX_POUNCE_MS = 400;
export const MAX_POUNCE_HEIGHT = 20;
export const MAX_STRETCH_MS = 1200;
export const BUBBLE_MS = 1000;
const ARRIVED_PX = 4;

const clamp = (v: number, lo: number, hi: number) => Math.min(hi, Math.max(lo, v));

export function createMax(x: number, rng: Rng): MaxState {
  return {
    action: 'sit',
    actionMs: 0,
    x,
    y: 0,
    facing: 1,
    jumpFrom: x,
    jumpTo: x,
    wanderTarget: null,
    wanderMs: between(rng, 3000, 6000),
    meowMs: between(rng, 20_000, 40_000),
    stretchMs: between(rng, 30_000, 60_000),
    bubble: null,
    bubbleMs: 0,
    bubbleCount: 0,
  };
}

function setAction(s: MaxState, action: MaxAction): MaxState {
  return { ...s, action, actionMs: 0 };
}

export function stepMax(prev: MaxState, input: MaxInput, dtMs: number, rng: Rng): MaxState {
  const s: MaxState = {
    ...prev,
    actionMs: prev.actionMs + dtMs,
    bubbleMs: prev.bubbleMs - dtMs,
    meowMs: prev.meowMs - dtMs,
  };
  if (s.action !== 'pounce') s.x = clamp(s.x, 0, input.width);
  if (s.bubbleMs <= 0) s.bubble = null;
  if (input.meow || s.meowMs <= 0) {
    s.bubbleCount += 1;
    s.bubble = { text: 'meow!', id: s.bubbleCount };
    s.bubbleMs = BUBBLE_MS;
    if (s.meowMs <= 0) s.meowMs = between(rng, 20_000, 40_000);
  }

  const busy = s.action === 'crouch' || s.action === 'pounce';
  if (input.pounceAt !== null && !input.reducedMotion && !busy) {
    s.jumpTo = clamp(input.pounceAt, 0, input.width);
    s.facing = s.jumpTo >= s.x ? 1 : -1;
    return setAction(s, 'crouch');
  }

  switch (s.action) {
    case 'crouch':
      if (s.actionMs < MAX_CROUCH_MS) return s;
      s.jumpFrom = s.x;
      return setAction(s, 'pounce');
    case 'pounce': {
      const p = Math.min(1, s.actionMs / MAX_POUNCE_MS);
      s.x = s.jumpFrom + (s.jumpTo - s.jumpFrom) * p;
      s.y = -4 * MAX_POUNCE_HEIGHT * p * (1 - p);
      if (p < 1) return s;
      s.y = 0;
      return setAction(s, 'sit');
    }
    case 'stretch':
      return s.actionMs >= MAX_STRETCH_MS ? setAction(s, 'sit') : s;
    default:
      return roam(s, input, dtMs, rng);
  }
}

function roam(s: MaxState, input: MaxInput, dtMs: number, rng: Rng): MaxState {
  if (input.target !== null) {
    s.wanderTarget = null;
  } else {
    s.wanderMs -= dtMs;
    if (s.wanderMs <= 0) {
      s.wanderTarget = rng() * input.width;
      s.wanderMs = between(rng, 3000, 6000);
    }
  }

  const goal = input.target ?? s.wanderTarget;
  const dist = goal === null ? 0 : clamp(goal, 0, input.width) - s.x;
  if (!input.reducedMotion && Math.abs(dist) > ARRIVED_PX) {
    const running = Math.abs(dist) > input.width / 4;
    const speed = running ? MAX_RUN_SPEED : MAX_WALK_SPEED;
    s.facing = dist > 0 ? 1 : -1;
    s.x += Math.sign(dist) * Math.min(Math.abs(dist), speed * dtMs);
    const action = running ? 'run' : 'walk';
    return s.action === action ? s : setAction(s, action);
  }

  s.stretchMs -= dtMs;
  if (s.stretchMs <= 0) {
    s.stretchMs = between(rng, 30_000, 60_000);
    return setAction(s, 'stretch');
  }
  return s.action === 'sit' ? s : setAction(s, 'sit');
}

export function viewMax(s: MaxState): CatView {
  return {
    pose: s.action,
    poseMs: s.actionMs,
    x: s.x,
    y: s.y,
    facing: s.facing,
    visible: true,
    bubble: s.bubble,
  };
}
```

```ts
import { between } from './anim';
import type { CatView, Pose, Rng } from './types';

export interface MisaInput {
  /** How far (px) she runs from her perch before she is out of sight. */
  exitDistance: number;
  /** A 1–3 line clear happened this step. */
  flinch: boolean;
  /** A Tetris happened this step. */
  tetris: boolean;
  /** The mouse cursor is close to her. */
  cursorNear: boolean;
  reducedMotion: boolean;
}

type MisaAction = 'peek' | 'flinch' | 'flee' | 'away' | 'return' | 'pause' | 'cower';

export interface MisaState {
  action: MisaAction;
  actionMs: number;
  /** Distance from her perch toward the exit, in px. */
  x: number;
  /** How long to stay away (or cower) before coming back. */
  awayMs: number;
  /** She pauses once on the way back. */
  paused: boolean;
}

export const MISA_FLINCH_MS = 600;
/** Speeds in px per ms. */
export const MISA_FLEE_SPEED = 0.4;
export const MISA_RETURN_SPEED = 0.05;
export const MISA_PAUSE_MS = 800;
export const MISA_CURSOR_AWAY_MS = 2000;

const SETTLED: readonly MisaAction[] = ['peek', 'flinch', 'return', 'pause'];

export function createMisa(): MisaState {
  return { action: 'peek', actionMs: 0, x: 0, awayMs: 0, paused: false };
}

function setAction(s: MisaState, action: MisaAction): MisaState {
  return { ...s, action, actionMs: 0 };
}

export function stepMisa(prev: MisaState, input: MisaInput, dtMs: number, rng: Rng): MisaState {
  let s: MisaState = { ...prev, actionMs: prev.actionMs + dtMs };

  const spooked = input.tetris || (input.cursorNear && SETTLED.includes(s.action));
  if (spooked) {
    const awayMs = input.tetris ? between(rng, 3000, 5000) : MISA_CURSOR_AWAY_MS;
    if (input.reducedMotion) return { ...setAction(s, 'cower'), awayMs };
    if (s.action === 'away') s = { ...s, actionMs: 0, awayMs: Math.max(s.awayMs, awayMs) };
    else if (s.action === 'flee') s = { ...s, awayMs: Math.max(s.awayMs, awayMs) };
    else return { ...setAction(s, 'flee'), awayMs, paused: false };
  }

  if (input.flinch && s.action === 'peek') return setAction(s, 'flinch');

  switch (s.action) {
    case 'flinch':
      return s.actionMs >= MISA_FLINCH_MS ? setAction(s, 'peek') : s;
    case 'flee':
      s.x += MISA_FLEE_SPEED * dtMs;
      if (s.x < input.exitDistance) return s;
      s.x = input.exitDistance;
      return setAction(s, 'away');
    case 'away':
    case 'cower':
      if (input.cursorNear) s.actionMs = 0; // wait until the cursor has gone
      if (s.actionMs < s.awayMs) return s;
      return s.action === 'cower' ? setAction(s, 'peek') : setAction(s, 'return');
    case 'return':
      s.x -= MISA_RETURN_SPEED * dtMs;
      if (s.x <= 0) return { ...setAction(s, 'peek'), x: 0 };
      if (!s.paused && s.x <= input.exitDistance / 2) return setAction(s, 'pause');
      return s;
    case 'pause':
      return s.actionMs >= MISA_PAUSE_MS ? { ...setAction(s, 'return'), paused: true } : s;
    default:
      return s;
  }
}

const POSES: Record<MisaAction, Pose> = {
  peek: 'peek',
  flinch: 'flinch',
  flee: 'run',
  away: 'run',
  return: 'walk',
  pause: 'sit',
  cower: 'flinch',
};

export function viewMisa(s: MisaState): CatView {
  return {
    pose: POSES[s.action],
    poseMs: s.actionMs,
    x: s.x,
    y: 0,
    facing: s.action === 'return' || s.action === 'pause' ? -1 : 1,
    visible: s.action !== 'away',
    bubble: null,
  };
}
```

```ts
import { between } from './anim';
import type { CatView, Rng } from './types';

export interface RayeInput {
  /** The stack is near the top (level). */
  danger: boolean;
  /** A Tetris happened this step. */
  tetris: boolean;
  /** The game is over (level). */
  over: boolean;
  /** Someone clicked her this step. */
  poked: boolean;
}

type RayeAction = 'sleep' | 'alert' | 'stretch' | 'yawn' | 'earTwitch' | 'oneEye';

export interface RayeState {
  action: RayeAction;
  actionMs: number;
  /** Time since danger ended, while alert. */
  calmMs: number;
  /** Time until her next stretch on her own. */
  stretchMs: number;
  /** Time until a click can wake her again. */
  cooldownMs: number;
}

export const RAYE_CALM_MS = 2000;
export const RAYE_TWITCH_MS = 400;
export const RAYE_STRETCH_MS = 1200;
export const RAYE_YAWN_MS = 800;
export const RAYE_POKE_COOLDOWN_MS = 4000;

export function createRaye(rng: Rng): RayeState {
  return {
    action: 'sleep',
    actionMs: 0,
    calmMs: 0,
    stretchMs: between(rng, 30_000, 60_000),
    cooldownMs: 0,
  };
}

function setAction(s: RayeState, action: RayeAction): RayeState {
  return { ...s, action, actionMs: 0 };
}

export function stepRaye(prev: RayeState, input: RayeInput, dtMs: number, rng: Rng): RayeState {
  const s: RayeState = {
    ...prev,
    actionMs: prev.actionMs + dtMs,
    cooldownMs: Math.max(0, prev.cooldownMs - dtMs),
  };

  if (input.over) return s.action === 'oneEye' ? s : setAction(s, 'oneEye');
  if (s.action === 'oneEye') return setAction(s, 'sleep');
  if (input.danger && s.action !== 'alert') return { ...setAction(s, 'alert'), calmMs: 0 };

  switch (s.action) {
    case 'alert':
      if (input.danger) return { ...s, calmMs: 0 };
      s.calmMs += dtMs;
      return s.calmMs >= RAYE_CALM_MS ? setAction(s, 'sleep') : s;
    case 'earTwitch':
      return s.actionMs >= RAYE_TWITCH_MS ? setAction(s, 'sleep') : s;
    case 'stretch':
      return s.actionMs >= RAYE_STRETCH_MS ? setAction(s, 'yawn') : s;
    case 'yawn':
      return s.actionMs >= RAYE_YAWN_MS ? setAction(s, 'sleep') : s;
    default: // sleep
      if (input.poked && s.cooldownMs <= 0) {
        return { ...setAction(s, 'stretch'), cooldownMs: RAYE_POKE_COOLDOWN_MS };
      }
      if (input.tetris) return setAction(s, 'earTwitch');
      s.stretchMs -= dtMs;
      if (s.stretchMs > 0) return s;
      return { ...setAction(s, 'stretch'), stretchMs: between(rng, 30_000, 60_000) };
  }
}

export function viewRaye(s: RayeState): CatView {
  return {
    pose: s.action,
    poseMs: s.actionMs,
    x: 0,
    y: 0,
    facing: 1,
    visible: true,
    bubble: s.action === 'sleep' ? { text: 'z z z', id: 0 } : null,
  };
}
```

```tsx
import { useEffect, useRef } from 'react';
import { BOARD_WIDTH } from '@mt/engine';
import { session } from '../net/sessionInstance';
import { frameIndex } from './anim';
import { centerColumn, dangerLevel, initialTracker, trackGame, type EventTracker } from './events';
import { createMax, stepMax, viewMax } from './max';
import { createMisa, stepMisa, viewMisa } from './misa';
import { createRaye, stepRaye, viewRaye } from './raye';
import { buildSheet, type Sheet } from './spriteSheet';
import { SCALE, SPRITE_SIZE } from './sprites';
import type { CatName, CatView } from './types';
import '../styles/cats.css';

export type CatScreen = 'menu' | 'game';

const CATS: readonly CatName[] = ['misa', 'max', 'raye'];
/** Max stops following a mouse that hasn't moved for this long. */
const CURSOR_IDLE_MS = 5000;
const MISA_NEAR_PX = 120;
/** Caps a frame's time step, e.g. after the tab was in the background. */
const MAX_DT_MS = 100;
/** Thickness of a panel's frame, so cats stand on it rather than inside it. */
const FRAME_PX = 5;

/** Where each cat lives this frame, in viewport px. */
interface Perches {
  /** Max walks along an edge from `left` to `left + width`, feet at `bottom`. */
  max: { left: number; bottom: number; width: number; cellPx: number | null };
  raye: { x: number; bottom: number };
  /** Misa's perch; she flees toward +x for `exitDistance` px. */
  misa: { x: number; bottom: number; exitDistance: number };
}

function rectOf(anchor: string): DOMRect | null {
  return document.querySelector(`[data-cat-anchor="${anchor}"]`)?.getBoundingClientRect() ?? null;
}

function findPerches(screen: CatScreen): Perches | null {
  if (screen === 'game') {
    const well = rectOf('well');
    const hold = rectOf('hold-left');
    const side = rectOf('side-right');
    if (!well || !hold || !side) return null;
    const misaX = side.left + side.width / 2;
    return {
      max: { left: well.left, bottom: well.top - FRAME_PX, width: well.width, cellPx: well.width / BOARD_WIDTH },
      raye: { x: hold.left + hold.width / 2, bottom: hold.top },
      misa: { x: misaX, bottom: side.bottom + 40, exitDistance: window.innerWidth - misaX + 40 },
    };
  }
  const menu = rectOf('menu');
  if (!menu) return null;
  return {
    // Max keeps clear of Raye's corner.
    max: { left: menu.left, bottom: menu.top, width: menu.width - 80, cellPx: null },
    raye: { x: menu.right - 40, bottom: menu.top },
    // She "runs behind the panel": a short dash toward it, then out of sight.
    misa: { x: menu.left - 20, bottom: menu.bottom - 16, exitDistance: 40 },
  };
}

export function CatLayer({ screen }: { screen: CatScreen }) {
  const screenRef = useRef(screen);
  const wrappers = useRef<Partial<Record<CatName, HTMLDivElement | null>>>({});
  const canvases = useRef<Partial<Record<CatName, HTMLCanvasElement | null>>>({});
  const bubbles = useRef<Partial<Record<CatName, HTMLDivElement | null>>>({});
  const poked = useRef(false);

  useEffect(() => {
    screenRef.current = screen;
  }, [screen]);

  useEffect(() => {
    const rng = Math.random;
    const reducedMotion = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches ?? false;
    const sheets: Record<CatName, Sheet> = {
      misa: buildSheet('misa'),
      max: buildSheet('max'),
      raye: buildSheet('raye'),
    };
    let max = createMax(0, rng);
    let misa = createMisa();
    let raye = createRaye(rng);
    let tracker: EventTracker = initialTracker;
    let danger = false;
    let cursor: { x: number; y: number; at: number } | null = null;
    const drawn: Record<CatName, string> = { misa: '', max: '', raye: '' };
    const bubbleIds: Record<CatName, number> = { misa: -1, max: -1, raye: -1 };

    const onMove = (e: MouseEvent) => {
      cursor = { x: e.clientX, y: e.clientY, at: performance.now() };
    };
    window.addEventListener('mousemove', onMove);

    const place = (name: CatName, view: CatView, originX: number, bottom: number) => {
      const wrap = wrappers.current[name];
      const canvas = canvases.current[name];
      const bubble = bubbles.current[name];
      if (!wrap || !canvas || !bubble) return;
      const size = SPRITE_SIZE * SCALE[name];
      wrap.style.visibility = view.visible ? 'visible' : 'hidden';
      wrap.style.transform = `translate(${Math.round(originX + view.x - size / 2)}px, ${Math.round(bottom + view.y - size)}px)`;
      canvas.style.transform = view.facing < 0 ? 'scaleX(-1)' : '';
      const frame = frameIndex(view.pose, view.poseMs);
      const key = `${view.pose}:${frame}`;
      if (drawn[name] !== key) {
        const ctx = canvas.getContext('2d');
        if (ctx) {
          ctx.clearRect(0, 0, SPRITE_SIZE, SPRITE_SIZE);
          ctx.drawImage(sheets[name][view.pose][frame], 0, 0);
        }
        drawn[name] = key;
      }
      if (view.bubble) {
        if (bubbleIds[name] !== view.bubble.id) {
          bubble.textContent = view.bubble.text;
          bubbleIds[name] = view.bubble.id;
        }
        bubble.classList.add('show');
      } else {
        bubble.classList.remove('show');
      }
    };

    let last = performance.now();
    let raf = 0;
    const frame = (now: number) => {
      raf = requestAnimationFrame(frame);
      const dt = Math.min(MAX_DT_MS, Math.max(0, now - last));
      last = now;
      const where = screenRef.current;
      const perches = findPerches(where);
      if (!perches) {
        for (const name of CATS) wrappers.current[name]?.style.setProperty('visibility', 'hidden');
        return;
      }

      const state = session.getState();
      const game = where === 'game' ? state.game : null;
      const tracked = trackGame(tracker, game);
      tracker = tracked.tracker;
      danger = dangerLevel(game, danger);
      let meow = false;
      let flinch = false;
      let tetris = false;
      let pounceAt: number | null = null;
      for (const event of tracked.events) {
        if (event.type === 'clear') {
          meow = true;
          if (event.count >= 4) tetris = true;
          else flinch = true;
        } else if (perches.max.cellPx !== null) {
          pounceAt = event.column * perches.max.cellPx;
        }
      }

      const active = game?.players[state.room?.you ?? 0].active ?? null;
      const recentCursor = cursor && now - cursor.at < CURSOR_IDLE_MS ? cursor : null;
      const maxTarget =
        perches.max.cellPx !== null
          ? active
            ? centerColumn(active) * perches.max.cellPx
            : null
          : recentCursor
            ? recentCursor.x - perches.max.left
            : null;
      const misaNear =
        where === 'menu' &&
        cursor !== null &&
        Math.hypot(cursor.x - perches.misa.x, cursor.y - (perches.misa.bottom - 16)) < MISA_NEAR_PX;

      max = stepMax(
        max,
        { width: perches.max.width, target: maxTarget, pounceAt, meow, reducedMotion },
        dt,
        rng,
      );
      misa = stepMisa(
        misa,
        { exitDistance: perches.misa.exitDistance, flinch, tetris, cursorNear: misaNear, reducedMotion },
        dt,
        rng,
      );
      raye = stepRaye(raye, { danger, tetris, over: game?.over ?? false, poked: poked.current }, dt, rng);
      poked.current = false;

      place('max', viewMax(max), perches.max.left, perches.max.bottom);
      place('misa', viewMisa(misa), perches.misa.x, perches.misa.bottom);
      place('raye', viewRaye(raye), perches.raye.x, perches.raye.bottom);
    };
    raf = requestAnimationFrame(frame);

    return () => {
      cancelAnimationFrame(raf);
      window.removeEventListener('mousemove', onMove);
    };
  }, []);

  return (
    <div className="cat-layer" aria-hidden="true">
      {CATS.map((name) => (
        <div
          key={name}
          className={`cat cat-${name}`}
          data-testid={`cat-${name}`}
          ref={(el) => {
            wrappers.current[name] = el;
          }}
        >
          <div
            className="cat-bubble"
            ref={(el) => {
              bubbles.current[name] = el;
            }}
          />
          <canvas
            width={SPRITE_SIZE}
            height={SPRITE_SIZE}
            style={{ width: SPRITE_SIZE * SCALE[name], height: SPRITE_SIZE * SCALE[name] }}
            ref={(el) => {
              canvases.current[name] = el;
            }}
            onClick={
              name === 'raye'
                ? () => {
                    poked.current = true;
                  }
                : undefined
            }
          />
        </div>
      ))}
    </div>
  );
}
```
